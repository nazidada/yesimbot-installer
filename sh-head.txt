#!/usr/bin/env bash
# install-yesimbot.sh 引导头 — 由 build-installers.mjs 拼接真实 mjs 内容
# 检查 node，提取内嵌 mjs，exec node 运行。
set -euo pipefail

# ---------- 1. 检查 node ----------
if ! command -v node >/dev/null 2>&1; then
  echo "[install] 未检测到 Node.js。请先安装 Node.js 18+ 后重新运行本脚本："
  echo "  macOS:   brew install node"
  echo "  Linux:   curl -fsSL https://deb.nodesource.com/setup_22.x | sudo -E bash - && sudo apt-get install -y nodejs"
  echo "  或访问 https://nodejs.org 下载安装包"
  echo "  或用 nvm: curl -o- https://raw.githubusercontent.com/nvm-sh/nvm/v0.40.3/install.sh | bash"
  exit 1
fi

# ---------- 2. 提取内嵌的 install-yesimbot.mjs ----------
TMPDIR="${TMPDIR:-/tmp}"
WORK_DIR="$(mktemp -d "${TMPDIR%/}/yesimbot-install.XXXXXX")"
MJS_FILE="${WORK_DIR}/install-yesimbot.mjs"

# 提取标记行之间的内容。python3 优先：用 rindex 取最后一次出现（真标记在文件尾部），
# 标记名用拼接避免匹配到提取代码自身。
if command -v python3 >/dev/null 2>&1; then
  python3 -c '
import sys
src = sys.stdin.read()
marker = "@@" + "MJS_BEGIN" + "@@"
endmarker = "@@" + "MJS_END" + "@@"
begin = src.rindex(marker)
end = src.rindex(endmarker)
sys.stdout.write(src[begin + len(marker) + 1:end])
' < "$0" > "${MJS_FILE}" || exit 1
else
  sed -n '/^@@MJS_BEGIN@@$/,/^@@MJS_END@@$/p' "$0" | sed '1d;$d' > "${MJS_FILE}" || exit 1
fi
# 防止文件被 CRLF 污染（Windows 编辑过）
tr -d '\r' < "${MJS_FILE}" > "${MJS_FILE}.clean" && mv "${MJS_FILE}.clean" "${MJS_FILE}"

# ---------- 3. 运行安装逻辑 ----------
# 删除临时目录（exit/信号时），再 exec node 替换当前进程
trap 'rm -rf "${WORK_DIR}"' EXIT INT TERM
exec node "${MJS_FILE}" "$@"

@@MJS_BEGIN@@
